=======================
ACL configuration on R1
=======================

ip access-list extended IT_INBOUND_ACL

permit tcp host 192.168.1.10 host 192.168.1.1 eq 23
permit tcp host 192.168.1.10 host 192.168.2.1 eq 23 

permit ip host 192.168.1.10 host 172.16.1.10
permit ip host 192.168.1.30 host 172.16.1.10

permit ip host 192.168.1.20 host 172.16.1.20

deny tcp 192.168.1.0 0.0.0.255 any eq telnet
deny ip 192.168.1.0 0.0.0.255 192.168.2.0 0.0.0.255

permit ip 192.168.1.0 0.0.0.255 any

interface ethernet 0/0 
 ip access-group IT_INBOUND_ACL in








=======================
ACL configuration on R2
=======================

ip access-list extended HR_INBOUND_ACL

deny ip 192.168.2.0 0.0.0.255 192.168.1.0 0.0.0.255

permit tcp 192.168.2.0 0.0.0.255 host 172.16.1.30 eq 80
permit tcp 192.168.2.0 0.0.0.255 host 172.16.1.30 eq 443
deny ip 192.168.2.0 0.0.0.255 any 

interface ethernet 0/0
 ip access-group HR_INBOUND_ACL in




ip access-list extended SERVER_INBOUND_ACL

deny icmp any 172.16.1.0  0.0.0.255
permit ip any any 

interface ethernet 0/2
 ip access-group SERVER_INBOUND_ACL out
