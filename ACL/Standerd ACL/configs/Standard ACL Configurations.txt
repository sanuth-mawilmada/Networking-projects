=============================
Standard ACL - Configurations
=============================


=================
R1 configurations
=================

username admin secret cisco
interface Loopback0
 ip address 1.1.1.1 255.255.255.255
interface Ethernet0/0
 ip address 192.168.1.1 255.255.255.0
interface Ethernet0/1
 ip address 10.10.10.1 255.255.255.252
interface Ethernet0/2
 no ip address
 shutdown 
interface Ethernet0/3
 no ip address
 shutdown
router ospf 1
 router-id 1.1.1.1
 network 1.1.1.1 0.0.0.0 area 0
 network 10.10.10.0 0.0.0.3 area 0
 network 192.168.1.0 0.0.0.255 area 0
ip forward-protocol nd
no ip http server
no ip http secure-server
control-plane
line con 0
 logging synchronous
line aux 0
line vty 0 4
 login local
 transport input telnet
end

=======================
ACL configuration on R1
=======================

access-list 10 permit host 192.168.1.10 

line vty 0 4
 access-class 10 in





access-list 20 permit host 192.168.2.1
access-list 20 deny 192.168.2.0 0.0.0.255
access-list 20 permit any 

interface ethernet 0/0
 ip access-group 20 out  








=======================
ACL configuration on R2
=======================

access-list 10 permit host 192.168.1.10

line vty 0 4 
 access-class 10 in 





access-list 20 permit 192.168.1.0 0.0.0.255
access-list 20 deny 192.168.2.0 0.0.0.255

interface ethernet 0/3
 ip access-group 20 out 





access-list 30 permit host 192.168.1.10
access-list 30 deny 192.168.1.0 0.0.0.255
access-list 30 permit any

interface  ethernet 0/0
 ip access-group 30 out 





access-list 40 permit host 192.168.1.10
access-list 40 permit host 192.168.1.30

interface ethernet 0/2
 ip access-group 40 out
