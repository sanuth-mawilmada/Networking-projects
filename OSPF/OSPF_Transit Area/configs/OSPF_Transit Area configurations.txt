================================
OSPF_Transit Area configurations
================================


=================
R1 configurations
=================


R1(config)#interface ethernet 0/0
R1(config-if)#no shutdown 
R1(config-if)#ip address 10.10.10.1 255.255.255.252


R1(config)#interface loopback 0
R1(config-if)#ip address 100.100.100.100 255.255.255.255


R1(config)#router ospf 1
R1(config-router)#router-id 100.100.100.100

R1(config-router)#network 100.100.100.100 0.0.0.0 area 0

R1(config-router)#network 10.10.10.1 0.0.0.3 area 1

R1(config-router)#area 1 virtual-link 200.200.200.200
